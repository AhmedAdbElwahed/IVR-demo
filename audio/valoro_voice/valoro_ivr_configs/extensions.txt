[from-internal]
exten => 1000,1,NoOp(Starting Valoro IVR)
 same => n,Goto(ivr-valoro,s,1)


[ivr-valoro]
exten => s,1,Answer()
 same => n,Set(CDR(userfield)=IVR_START)
 same => n,Set(CALLER_PHONE=${CALLERID(num)})
 same => n,Set(CALLER_NAME=${CALLERID(name)})
 same => n,Playback(valoro/valoro-welcome)
 same => n,Playback(valoro/valoro-main-menu)
 same => n,WaitExten(10)

; 1 → Sales
exten => 1,1,Goto(ivr-sales,s,1)

; 2 → Support
exten => 2,1,Goto(ivr-support,s,1)

; 3 → Billing
exten => 3,1,Goto(ivr-billing,s,1)

; 0 → Operator (always)
exten => 0,1,Goto(ivr-operator,s,1)

; Invalid
exten => i,1,Playback(valoro-invalid)
 same => n,Goto(ivr-valoro,s,2)

; Timeout
exten => t,1,Playback(valoro-timeout)
 same => n,Goto(ivr-valoro,s,2)


[ivr-sales]
exten => s,1,NoOp(Sales Menu)
 same => n,Playback(valoro/valoro-sales-menu)
 same => n,WaitExten(10)

; Pricing inquiry

 same => n,Playback(valoro/valoro-sales-pricing)

; Callback request
exten => 2,1,Set(DEPT=sales)
 same => n,Set(TICKET_TYPE=callback_request)

; Operator
exten => 0,1,Goto(ivr-operator,s,1)

; Invalid
exten => i,1,Playback(valoro/valoro-invalid)
 same => n,Goto(ivr-sales,s,1)

; Timeout
exten => t,1,Playback(valoro/valoro-timeout)
 same => n,Goto(ivr-sales,s,1)
